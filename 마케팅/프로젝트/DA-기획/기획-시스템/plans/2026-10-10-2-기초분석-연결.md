```python
START_RE = re.compile(r"(\d{4})\.\s*(\d{1,2})\.\s*(\d{1,2})\.")


def parse_start_date(line):
    """'2026. 5. 3.에 게재 시작함' → date(2026, 5, 3). 못 읽으면 None."""
    if not line:
        return None
    m = START_RE.search(line)
    if not m:
        return None
    try:
        return dt.date(int(m.group(1)), int(m.group(2)), int(m.group(3)))
    except ValueError:
        return None


def long_running(analyses, asof, min_days=30, top=20):
    """활성 광고를 게재일수 내림차순으로. min_days 미만은 빼고, 같은 라이브러리 ID는 한 번만.
    한국 광고 라이브러리는 꺼진 상업 광고를 보여주지 않으므로 게재일수 = 기준일 - 시작일이다."""
    rows, seen = [], set()
    for a in analyses:
        for ad in a.get("ads", []):
            if ad.get("status") not in ("활성", None):
                continue
            lid = ad.get("library_id")
            if lid and lid in seen:
                continue
            d = parse_start_date(ad.get("start_date"))
            if not d:
                continue
            days = (asof - d).days
            if days < min_days:
                continue
            if lid:
                seen.add(lid)
            rows.append({"brand": a["brand"], "page_name": ad.get("page_name") or "-",
                         "library_id": lid or "-", "start": d.isoformat(), "days": days,
                         "hook": extract_hooks(ad.get("body") or "").replace("|", "/")})
    rows.sort(key=lambda r: r["days"], reverse=True)
    return rows[:top]
```

```python
def load_targets_from_keywords(text):
    """결핍 키워드 목록('주사피부염, 홍조') → 브랜드 목록과 같은 모양의 검색 대상."""
    seen, out = set(), []
    for k in re.split(r"[,\n]", text):
        k = k.strip()
        if not k or k in seen:
            continue
        seen.add(k)
        out.append({"brand": k, "keywords": [k], "total_volume": 0, "category": "결핍"})
```

```python
"""
메타 광고 라이브러리에서 특정 광고 ID의 영상 1편을 받는다 (Gemini 영상 분석용).

?id=<ID>로 열면 그 광고주의 광고 전체가 뜬다. 'ID 표시가 있는 카드'에서 위로 올라가며
그 카드 안의 video 주소를 찾는다(라이브러리 ID가 둘 이상 든 상자에 닿으면 멈춘다).

사용:
    python fetch_ad_video.py --ids "1498055228605417,952907087430431" --out <폴더>
결과: <폴더>/<ID>.mp4, <폴더>/_fetch.json (ID별 성공·실패)
"""
import argparse, json, re, time
from pathlib import Path

import requests
from playwright.sync_api import sync_playwright

FIND_JS = r"""(id) => {
  const m = [...document.querySelectorAll('*')].find(el => el.children.length === 0 && (el.textContent || '').includes('라이브러리 ID: ' + id));
  if (!m) return {marker: false, src: null};
  let c = m;
  for (let i = 0; i < 15 && c; i++) {
    c = c.parentElement;
    if (!c) break;
    if ((c.innerText.match(/라이브러리 ID/g) || []).length > 1) break;
    const v = c.querySelector('video');
    if (v) return {marker: true, src: v.src || v.currentSrc || null};
  }
  return {marker: true, src: null};
}"""


def parse_ids(text):
    seen, out = set(), []
    for tok in re.split(r"[,\s]+", text):
        if tok.isdigit() and tok not in seen:
            seen.add(tok)
            out.append(tok)
    return out


def ad_url(lid):
    return f"https://www.facebook.com/ads/library/?id={lid}"


def fetch(ids, out_dir, headless=True):
    out_dir.mkdir(parents=True, exist_ok=True)
    sess = requests.Session()
    sess.headers.update({"User-Agent": "Mozilla/5.0", "Referer": "https://www.facebook.com/"})
    results = []
    with sync_playwright() as p:
        browser = p.chromium.launch(headless=headless)
        page = browser.new_context(locale="ko-KR", viewport={"width": 1366, "height": 900}).new_page()
        for lid in ids:
            r = {"id": lid, "url": ad_url(lid), "file": None, "error": None}
            try:
                page.goto(ad_url(lid), timeout=60000, wait_until="domcontentloaded")
                page.wait_for_timeout(6000)
                found = page.evaluate(FIND_JS, lid)
                if not found["marker"]:
                    r["error"] = "카드 없음(광고가 꺼졌거나 ID 오류)"
                elif not found["src"]:
                    r["error"] = "영상 없음(이미지 광고일 수 있음)"
                else:
                    resp = sess.get(found["src"], timeout=120, stream=True)
                    if resp.status_code != 200:
                        r["error"] = f"다운로드 HTTP {resp.status_code}"
                    else:
                        f = out_dir / f"{lid}.mp4"
                        with open(f, "wb") as fh:
                            for chunk in resp.iter_content(65536):
                                fh.write(chunk)
                        r["file"] = f.name
            except Exception as e:
                r["error"] = str(e)[:200]
            print(f"  {lid}: {r['file'] or r['error']}", flush=True)
            results.append(r)
            time.sleep(1)
        browser.close()
    (out_dir / "_fetch.json").write_text(json.dumps(results, ensure_ascii=False, indent=2), encoding="utf-8")
    return results


if __name__ == "__main__":
    ap = argparse.ArgumentParser(description="광고 ID로 메타 광고 영상 받기")
    ap.add_argument("--ids", required=True)
    ap.add_argument("--out", required=True)
    ap.add_argument("--no-headless", action="store_true")
    a = ap.parse_args()
    fetch(parse_ids(a.ids), Path(a.out), headless=not a.no_headless)
```
